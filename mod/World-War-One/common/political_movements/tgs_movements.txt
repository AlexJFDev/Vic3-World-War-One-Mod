movement_pacifist = {
	category = movement_category_ideological

	ideology = ideology_pacifist
	character_ideologies = {
		ideology_pacifist
	}

	can_cause_obstinance = yes

	disband_trigger = {
		OR = {
            has_law = law_type:law_true_neutrality
			has_law = law_type:law_armed_neutrality
		}
	}
	on_disbanded = {}

	creation_trigger = {
		has_technology_researched = human_rights
		OR = {
			has_law = law_type:law_limited_interventionism
			has_law = law_type:law_expansionism
			has_law = law_type:law_imperialism
		}
	}
	creation_weight = {
		value = 100
	}
	on_created = {
		add_movement_enthusiasm_modifier = yes
	}

	character_support_trigger = {
		culture = {
			is_primary_culture_of = root.owner
		}
		OR = {
			has_ideology = ideology:ideology_pacifist
			has_ideology = ideology:ideology_utopian
			has_ideology = ideology:ideology_humanitarian
			has_ideology = ideology:ideology_humanitarian_royalist
		}
	}
	character_support_weight = {
		value = 100
		if = {
			limit = {
				has_ideology = ideology:ideology_pacifist
			}
			multiply = {
				value = 5
			}
		}
		else_if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_utopian
				}
			}
			multiply = {
				value = 5
			}
		}
		else_if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_humanitarian
				}
			}
			multiply = {
				value = 5
			}
		}
		else_if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_humanitarian_royalist
				}
			}
			multiply = {
				value = 5
			}
		}
	}

	pop_support_trigger = {
		culture = {
			is_primary_culture_of = root.owner
		}
	}

	pop_support_factors = {
		movement_support_high_literacy
		movement_support_farmers
		movement_support_clerks
		movement_support_peasants
		movement_support_academics
		movement_support_below_expected_sol
	}

	pop_support_weight = {
		if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= {
						OR = {
							is_being_suppressed = yes
							is_being_bolstered = yes
						}
					}
				}
			if = {
				limit = {
					scope:political_movement = {
						is_being_suppressed = yes
					}
				}
				add = {
					value = suppression_base_value
					multiply = {
						value = -0.5
						add = owner.modifier:country_suppression_attraction_factor
					}
					desc = "concept_suppressed"
				}
			}
			else = {
				add = {
					value = bolstering_base_value
					multiply = {
						value = 0.5
						add = owner.modifier:country_bolster_attraction_factor
					}
					desc = "concept_bolstered"
				}
			}
		}

		add = {
			value = 3
			desc = "POP_BASE_SUPPORT"
		}

		add = {
			desc = "URBAN_STATE"
			value = 10
			multiply = state.state_urbanization_rate
		}

		if = {
			limit = {
				strata = upper
			}
			if = {
				limit = {
					is_pop_type = capitalists
				}
				add = {
					value = 50
					desc = "POP_CAPITALISTS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = aristocrats
				}
				if = {
					limit = {
						culture = cu:hungarian
						owner ?= {
							has_journal_entry = je_hungarian_revolution
						}
					}
					add = {
						value = 25
						desc = "je_hungarian_revolution"
					}
				}
				else = {
				add = {
					value = 6
					desc = "POP_ARISTOCRATS"
				}
			}
			}
			else = {
				add = {
					value = 9
					desc = "UPPER_NO_ICON"
				}
			}
		}
		else_if = {
			limit = {
				strata = middle
			}
			if = {
				limit = {
					is_pop_type = shopkeepers
				}
				add = {
					value = 20
					desc = "POP_SHOPKEEPERS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = academics
				}
				add = {
					value = 35
					desc = "POP_ACADEMICS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = bureaucrats
				}
				add = {
					value = 15
					desc = "POP_BUREAUCRATS"
				}
			}
			else = {
				add = {
					value = 13
					desc = "MIDDLE_NO_ICON"
				}
			}
		}
		else_if = {
			limit = {
				strata = lower
			}
			if = {
				limit = {
					is_pop_type = clerks
				}
				add = {
					value = 9
					desc = "POP_CLERKS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = peasants
				}
				add = {
					value = 0
					desc = "POP_PEASANTS"
				}
			}
			else = {
				add = {
					value = 3
					desc = "LOWER_NO_ICON"
				}
			}
		}

		multiply = { # Illiterate pops participate at 1/10 the rate of fully literate pops
			desc = "POP_LITERACY"
			value = literacy_rate
			add = 0.10
		}
	}

	revolution = {
		possible = {
			political_movement_support >= 0.05
		}

		weight = {
			value = 100
		}

		state_weight = {
			value = 1

			if = {
				limit = {
					is_capital = no
					is_homeland_of_country_cultures = owner
					primary_cultures_percent_state >= 0.05
				}
				add = 10
			}
		}
	}

	secession = {
		possible = {
			political_movement_support >= 0.05
			scope:clout <= 0
		}

		weight = {
			value = 1000
		}

		state_weight = {
			value = 0

			if = {
				limit = {
					is_capital = no
					is_homeland_of_country_cultures = owner
					primary_cultures_percent_state >= 0.05
				}
				add = 10
			}
		}
	}
}

	category = movement_category_ideological

	ideology = ideology_tenentist

	}

	can_cause_obstinance = yes

	creation_trigger = {

	}

	}

	character_support_trigger = {

			is_primary_culture_of = root.owner
		}
		OR = {
			has_ideology = ideology:ideology_tenentist
			has_ideology = ideology:ideology_land_reformer
			has_ideology = ideology:ideology_modernizer_leader
			has_ideology = ideology:ideology_fascist
			has_ideology = ideology:ideology_integralist
		}
	}
	character_support_weight = {
		value = 100
		if = {
			limit = {
				has_ideology = ideology:ideology_tenentist
			}
			multiply = {
				value = 5
			}
		}
		else_if = {
			limit = {
				OR = {
					has_ideology = ideology_land_reformer
				}
			}
			multiply = {
				value = 5
			}
		}
		else_if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_modernizer_leader
				}
			}
			multiply = {
				value = 5
			}
		}
		else_if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_fascist
				}
			}
			multiply = {
				value = 5
			}
		}
		else_if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_integralist
				}
			}
			multiply = {
				value = 5
			}
		}
	}

	pop_support_trigger = {
		culture = {
			is_primary_culture_of = root.owner
		}
	}

	pop_support_factors = {
		movement_support_officers
		movement_support_soldiers_conscripts
		movement_support_high_literacy
		movement_support_farmers
		movement_support_clerks
		movement_support_peasants
		movement_support_academics
		movement_support_below_expected_sol
	}

	pop_support_weight = {
		if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= {
						OR = {
							is_being_suppressed = yes
							is_being_bolstered = yes
						}
					}
				}
			if = {
				limit = {
					scope:political_movement = {
						is_being_suppressed = yes
					}
				}
				add = {
					value = suppression_base_value
					multiply = {
						value = -0.5
						add = owner.modifier:country_suppression_attraction_factor
					}
					desc = "concept_suppressed"
				}
			}
			else = {
				add = {
					value = bolstering_base_value
					multiply = {
						value = 0.5
						add = owner.modifier:country_bolster_attraction_factor
					}
					desc = "concept_bolstered"
				}
			}
		}
		add = {
			value = 1
			desc = "POP_BASE_SUPPORT"
		}

		if = {
			limit = {
				strata = upper
			}
			if = {
				limit = {
					is_pop_type = aristocrats
				}
				add = {
					value = 18
					desc = "POP_ARISTOCRATS"
				}
			}
			else = {
				add = {
					value = 6
					desc = "UPPER_NO_ICON"
				}
			}
			if = {
				limit = {
					owner ?= {
						any_political_movement = {
							OR = {
								is_political_movement_type = movement_socialist
								is_political_movement_type = movement_anarchist
								is_political_movement_type = movement_communist
							}
							political_movement_support >= 0.10 # (chuckles) im in danger
						}
					}
				}
				multiply = {
					value = 1.5
					desc = "STRONG_SOCIALIST_MOVEMENT"
				}
			}
		}
		else_if = {
			limit = {
				strata = middle
			}
			if = {
				limit = {
					is_pop_type = clergymen
				}
				add = {
					value = 25
					desc = "POP_CLERGYMEN"
				}
			}
			else_if = {
				limit = {
					is_pop_type = farmers
				}
				add = {
					value = 15
					desc = "POP_FARMERS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = officers
				}
				add = {
					value = 12
					desc = "POP_SHOPKEEPERS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = shopkeepers
				}
				add = {
					value = 10
					desc = "POP_SHOPKEEPERS"
				}
			}
			else = {
				add = {
					value = 9
					desc = "MIDDLE_NO_ICON"
				}
			}

			if = {
				limit = {
					owner ?= {
						any_political_movement = {
							OR = {
								is_political_movement_type = movement_socialist
								is_political_movement_type = movement_anarchist
								is_political_movement_type = movement_communist
							}
							political_movement_support >= 0.10
						}
					}
				}
				multiply = {
					value = 1.25
					desc = "STRONG_SOCIALIST_MOVEMENT"
				}
			}
		}
		else_if = {
			limit = {
				strata = lower
			}
			if = {
				limit = {
					is_pop_type = soldiers
				}
				add = {
					value = 6
					desc = "POP_CLERKS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = peasants
				}
				add = {
					value = 15
					desc = "POP_PEASANTS"
				}
			}
			else = {
				add = {
					value = 3
					desc = "LOWER_NO_ICON"
				}
			}
		}

		if = {
			limit = {
				standard_of_living < state.average_expected_sol
				strata = middle
			}
			add = { # Increases likelihood of support if impoverished
				desc = "IS_BELOW_EXPECTED_SOL"
				value = state.average_expected_sol
				subtract = standard_of_living
				max = 10
			}
		}
		else_if = {
			limit = {
				standard_of_living > state.average_expected_sol
				strata = middle
			}
			add = { # Decreases likelihood of support if well-off
				desc = "IS_ABOVE_EXPECTED_SOL"
				value = state.average_expected_sol
				subtract = standard_of_living
				min = -5
			}
		}

		if = {
			limit = {
				is_employed = no
			}
			multiply = {
				value = 1.5
				desc = "UNEMPLOYED"
			}
		}

		multiply = { # Illiterate pops participate at 1/10 the rate of fully literate pops
			desc = "POP_LITERACY"
			value = literacy_rate
			add = 0.10
		}

		multiply = { # Intentional that this is different from fascism
			value = 0.01
			multiply = pop_acceptance
			add = 0.20
			desc = "ACCEPTANCE"
		}
	}

	revolution = {
		possible = {
			political_movement_support >= 0.05
		}

		weight = {
			value = 100
		}

		state_weight = {
			value = 1

			if = {
				limit = {
					is_capital = no
					is_homeland_of_country_cultures = owner
					primary_cultures_percent_state >= 0.05
				}
				add = 10
			}
		}
	}

	secession = {
		possible = {
			political_movement_support >= 0.05
			scope:clout <= 0
		}

		weight = {
			value = 1000
		}

		state_weight = {
			value = 0

			if = {
				limit = {
					is_capital = no
					is_homeland_of_country_cultures = owner
					primary_cultures_percent_state >= 0.05
				}
				add = 10
			}
		}
	}
}
