je_fascist_threat = {
    icon = "gfx/interface/icons/event_icons/italian_fascism.dds"
    group = je_group_internal_affairs

    is_shown_when_inactive = {
        exists = c:ITA
        THIS = c:ITA
    }
    
    possible = {
        has_global_variable = { italy_mutilated_victory }
        NOR = {
			has_law = law_type:law_council_republic
			has_law = law_type:law_anarchy
		}
        any_country = {
			limit = {
                country_rank >= rank_value:great_power
                OR = {
                    has_law = law_type:law_council_republic
                    has_law = law_type:law_anarchy
                    has_law = law_type:law_cooperative_ownership
                }
            }
        }  
    }

    immediate = {
        trigger_event = {
            id = italy_events.10
            days = 5
        }
        trigger_event = {
            id = italy_events.11
            days = 30
        }
    }

    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = je_fascist_threat_completion_desc
        trigger = {
            game_date > 1922.10.27
            OR = {
                NOT = {
                    any_political_movement = {
				       is_political_movement_type = movement_fascist
			        }
                }
                any_political_movement = {
				   is_political_movement_type = movement_fascist
                   political_movement_support < 0.05
			    }
            }
        }
        effect = { set_variable = avoided_fascist_movement }
    }

    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = je_fascist_threat_failure_desc
        trigger = {
            game_date > 1922.10.27
            any_political_movement = {
				is_political_movement_type = movement_fascist
                political_movement_support > 0.2
			}
        }
        effect = {
            trigger_event = {
              id = italy_events.13
              days = 5
            }
        }
    }

    outcome = {
        flags = { hidden invalid_outcome }
        conclusion_desc = je_standard_failure_conclusion_desc
        trigger = { 
            OR = {
				has_law = law_type:law_council_republic
				has_law = law_type:law_anarchy
			}
        }
        effect = { set_variable = avoided_fascist_movement }
    }

    weight = 1000
    highlight = yes
    transferable = no
    can_revolution_inherit = no
    should_be_pinned_by_default_uninvolved_or_context = yes
}

	}

		}
