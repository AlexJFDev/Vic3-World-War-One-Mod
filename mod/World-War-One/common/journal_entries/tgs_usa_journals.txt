                c:GER = { is_diplomatic_play_enemy_of = c:FRA }
                c:GER = { is_diplomatic_play_enemy_of = c:RUS }
            }
            NOT = {
              has_variable = entente_participant
              has_variable = central_powers_participant
            }
        }
        effect = {
            trigger_event = {
              id = usa_events.20
              days = 5
            }
        }
        event_outcome_desc = {} # complex desc like a status_desc field
    }

    outcome = {
        header = je_standard_failure_header
        effect_header = je_us_neutrality_failure_desc

        conclusion_desc = je_standard_failure_conclusion_desc
        trigger = {

              has_variable = entente_participant
              has_variable = central_powers_participant
