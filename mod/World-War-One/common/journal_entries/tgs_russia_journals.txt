je_petrograd_soviet = {
    icon = "gfx/interface/icons/event_icons/petrograd_soviet.dds"
    group = je_group_internal_affairs

    activation_description = {
		always = yes
	}

    is_shown_when_inactive = {
        exists = c:RUS
        THIS = c:RUS
        has_global_variable = russian_revolution_1
    }
    
    possible = {
        exists = c:RUS
        THIS = c:RUS
        has_global_variable = russian_revolution_1
    }

    modifiers_while_active = { petrograd_soviet }
 
    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = je_petrograd_soviet_completion_desc
        trigger = {
            OR = {
                root.average_sol >= 10
                has_law = law_type:law_council_republic
                ig:ig_trade_unions ?= {
				   ig_approval >= loyal
			    }
            }
        }
        effect = {
            random_civil_war = {
			  add_civil_war_progress = -100
		    }
        }
        event_outcome_desc = {} # complex desc like a status_desc field
    }

    outcome = {
        header = je_standard_failure_header
        effect_header = je_petrograd_soviet_failure_desc
        color = "blue_negative"
        conclusion_desc = je_standard_failure_conclusion_desc
        trigger = {
            OR = {
                any_civil_war = {
                    this.political_movement ?= {
                        any_influenced_interest_group = {
                            law_stance = {
                              law = law_type:law_council_republic
                              value > neutral
                            }
                        }
                    }
                    is_civil_war_type = revolution
                    civil_war_progress >= 0.75
                }
                AND = {
                    is_at_war = yes
                    NOT = { has_global_variable = world_war_1_over }
                    has_variable = russian_rep_continues_war
                }
            }
        }
        effect = {
            random_civil_war = {
			  add_civil_war_progress = -100
		    }
            trigger_event = {
              id = russia_events.7
              days = 5
            }
        }
    }

    weight = 1000
    highlight = yes
    transferable = no
    can_revolution_inherit = yes
    should_be_pinned_by_default_uninvolved_or_context = yes
}
