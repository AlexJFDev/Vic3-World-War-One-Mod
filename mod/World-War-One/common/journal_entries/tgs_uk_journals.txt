je_irish_question = {
    icon = "gfx/interface/icons/event_icons/irish_rep_flag.dds"
    group = je_group_internal_affairs

    activation_description = {
		always = yes
	}

    is_shown_when_inactive = {
        exists = c:GBR
        THIS = c:GBR
    }
    
    possible = {
		owns_entire_state_region = STATE_LEINSTER
        owns_entire_state_region = STATE_ULSTER
        owns_entire_state_region = STATE_MUNSTER
        owns_entire_state_region = STATE_CONNAUGHT
    }

    immediate = {
        trigger_event = { 
           id = uk_events.6
           days = 5
        }
    }

    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = je_irish_question_failure_desc
        trigger = {
            any_political_movement = {
			   is_political_movement_type = movement_cultural_minority
               culture ?= cu:irish
               is_insurrectionary = yes
			   civil_war ?= {
                    OR = {
                       is_civil_war_type = revolution
                       is_civil_war_type = secession
                    }
				    civil_war_progress >= 0.75
			    }
            }
        }
        effect = {
            trigger_event = { 
              id = uk_events.8
              days = 5
            }
        }
    }

    outcome = {
        flags = { hidden invalid_outcome }
        conclusion_desc = je_irish_question_failure_desc
        trigger = {
            NOT = {
                owns_entire_state_region = STATE_LEINSTER
                owns_entire_state_region = STATE_ULSTER
                owns_entire_state_region = STATE_MUNSTER
                owns_entire_state_region = STATE_CONNAUGHT
            }
        }
        effect = {
            set_variable = alt_independant_ireland
        }
    }

    weight = 1000
    highlight = yes
    transferable = yes
    can_revolution_inherit = yes
    should_be_pinned_by_default_uninvolved_or_context = yes
}
