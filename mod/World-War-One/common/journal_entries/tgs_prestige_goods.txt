je_prestige_goods_movies = {
    icon = "gfx/interface/icons/goods_icons/prestige_goods/prestige_good_generic_movies.dds"

    group = je_group_technology

    activation_description = {
		always = yes
	}

    is_shown_when_inactive = {
        has_dlc_feature = mp1_content
        any_company = {
            can_potentially_produce_prestige_goods = prestige_good_generic_movies
            is_producing_prestige_goods = no
            NOT = {
                has_variable = prestige_good_generic_movies_var
            }
        }
    }

    possible = {
        any_company = {
            can_potentially_produce_prestige_goods = prestige_good_generic_movies
            is_producing_prestige_goods = no
            company_is_prosperous = yes
            NOT = {
                custom_tooltip = {
                    text = not_prestige_good_generic_movies_var_tt
                    has_variable = prestige_good_generic_movies_var
                }
            }
        }
    }

    immediate = {
        random_company = {
            limit = {
                can_potentially_produce_prestige_goods = prestige_good_generic_movies
                is_producing_prestige_goods = no
                company_is_prosperous = yes
                NOT = {
                    has_variable = prestige_good_generic_movies_var
                }
            }
            save_scope_as = relevant_company
        }
        set_variable = {
            name = prestige_good_generic_movies_var
            value = 0
        }
        g:popular_art = {
            save_scope_as = relevant_goods
        }
    }

    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = je_standard_completion_conclusion_desc
        trigger = {
        trigger_if = {
            limit = {
                has_variable = prestige_good_generic_movies_var
            }
            scope:journal_entry = {
                is_goal_complete = yes
            }
        }
        trigger_if = {
            limit = {
                exists = scope:relevant_company
            }
            scope:relevant_company ?= {
                company_is_prosperous = yes
            }
        }
        trigger_if = {
            limit = {
                exists = scope:relevant_goods
            }
            custom_tooltip = {
                text = relevant_goods_production_rank_tt
                "goods_production_rank(scope:relevant_goods)" <= 3
            }
        }
        }
        effect = {
        if = {
            limit = {
                exists = scope:relevant_company
            }
            custom_tooltip = {
                text = je_prestige_goods_popular_art_reward_tt
                scope:relevant_company ?= {
                    set_variable = prestige_good_generic_movies_var
                }
            }
        }
        post_notification = prestige_goods_popular_art_unlocked
        }
    }

    on_monthly_pulse = {
        effect = {
            if = {
                limit = {
                    scope:relevant_company ?= {
                        company_is_prosperous = yes
                    }
                    "goods_production_rank(scope:relevant_goods)" <= 3
                }
                change_variable = { name = prestige_good_generic_movies_var add = 1 }
            }
        }
    }

    current_value = {
        value = root.var:prestige_good_generic_movies_var
    }

    goal_add_value = {
        add = prestige_good_journal_entry_months
    }

    progressbar = yes

    weight = 1000

    highlight = yes
    transferable = no
    can_revolution_inherit = no
    should_be_pinned_by_default_uninvolved_or_context = yes
}
