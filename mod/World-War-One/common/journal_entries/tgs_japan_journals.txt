        game_date > 1925.1.1

        conclusion_desc = je_japan_expansion_completion_desc
        trigger = {
            OR = {
                c:MCH = {
                    NOT = { 
                      is_subject_of = c:JAP
                    }
                    OR = {
                      is_subject = no
                      is_subject_of = c:CHI
                    }
                }
                c:CHI = {
                  has_state_in_state_region = STATE_NORTHERN_MANCHURIA
				  has_state_in_state_region = STATE_SHENGJING
				  has_state_in_state_region = STATE_SOUTHERN_MANCHURIA
                }
            }
            NOT = {
                c:JAP = {
                    any_scope_treaty = {
					    binds = c:MCH
					    any_scope_article = {
					       OR = {
						     has_type = alliance
						     has_type = defensive_pact
						    }
					    }
				    }

            game_date > 1931.9.1

                OR = {
				  has_ideology = ideology:ideology_fascist
